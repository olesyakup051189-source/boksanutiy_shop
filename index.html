    function parseDriveImageUrl(val) {

      if (!val) return '';

      const trimmed = val.trim();

      const match = trimmed.match(/[-\w]{25,}/);

      if (trimmed.includes('drive.google.com') && match) {

        return `https://drive.google.com/thumbnail?id=${match[0]}&sz=w800`; } if (/^[-\w]{25,}$/.test(trimmed)) { return `https://drive.google.com/thumbnail?id=${trimmed}&sz=w800`;

      }

      if (trimmed.startsWith('http')) return trimmed;

      return '';

    }


    async function loadFromGoogleSheet() {

      const url = `https://docs.google.com/spreadsheets/d/${SHOP_CONFIG.SHEET_ID}/gviz/tq?tqx=out:json&sheet=${encodeURIComponent(SHOP_CONFIG.SHEET_NAME)}`;

      try {

        const res = await fetch(url);

        const text = await res.text();

        const json = JSON.parse(text.substring(47).slice(0, -2));

        const rows = json.table.rows;


        const parsed = [];

        rows.forEach((r, idx) => {

          const c = r.c;

          if (!c || !c[1] || !c[1].v) return;

          const name = String(c[1].v).trim();

          if (name === 'Товар') return;


          const priceRaw = c[3] ? (c[3].v || c[3].f || '0') : '0';

          const price = Number(String(priceRaw).replace(/[^\d]/g, '')) || 0;

          const stock = c[5] && c[5].v !== null ? Number(c[5].v) : 1;

          const desc = c[6] && c[6].v ? String(c[6].v) : '';

          const photo = c[7] && c[7].v ? parseDriveImageUrl(String(c[7].v)) : '';

          const category = c[8] && c[8].v ? String(c[8].v).trim() : 'Экипировка';


          parsed.push({ id: idx + 1, name, price, stock, desc, photo, category });

        });


        if (parsed.length > 0) {

          const inStockNames = new Set(parsed.filter(p => p.stock > 0).map(p => p.name));

          products = parsed.filter(p => p.stock > 0 || !inStockNames.has(p.name));

          renderFilters();

          renderCatalog();

        }

      } catch (e) {

        console.warn('Используется резервный каталог:', e);

      }

    }

      container.innerHTML = categories.map(cat => `

        <button class="filter-btn ${cat === activeCategory ? 'active' : ''}" onclick="setCategory('${cat}')">${cat}</button>

      `).join('');

    }


    function setCategory(cat) {

      activeCategory = cat;

      renderFilters();

      renderCatalog();

    }


    function renderCatalog() {

      const query = document.getElementById('search-input').value.toLowerCase();

      const grid = document.getElementById('catalog-grid');


      const filtered = products.filter(p => {

        const matchCat = activeCategory === 'Все' || p.category === activeCategory;

        const matchSearch = p.name.toLowerCase().includes(query) || p.desc.toLowerCase().includes(query);

        return matchCat && matchSearch;

      });


      grid.innerHTML = filtered.map(p => `

        <article class="card">

          <div class="card-img-wrap" onclick="openProduct(${p.id})">

            <span class="badge">${p.category}</span>

            <span class="stock-badge">${p.stock > 0 ? `В наличии: ${p.stock} шт.` : 'Нет в наличии'}</span>

            ${p.photo

              ? `<img src="${p.photo}" alt="${p.name}" loading="lazy" onerror="this.style.display='none'; this.nextElementSibling.style.display='block';" /><div class="img-placeholder" style="display:none;">WINNING / WINDY</div>`

              : `<div class="img-placeholder">WINNING<br><span style="font-size:12px; color:#9a9894;">ORIGINAL EQUIPMENT</span></div>`

            }

          </div>

          <div class="card-body">

            <h3 class="card-title" onclick="openProduct(${p.id})">${p.name}</h3>

            <p class="card-desc-short">${p.desc}</p>

            <div class="card-footer">

              <div class="price">${p.price.toLocaleString('ru-RU')} ¥</div>

              <button class="btn-add" ${p.stock <= 0 ? 'disabled' : ''} onclick="addToCart(${p.id})">

                ${p.stock > 0 ? 'В корзину' : 'Продано'}

              </button>

            </div>

          </div>

        </article>

      `).join('');

    }


    function openProduct(id) {

      const p = products.find(item => item.id === id);

      if (!p) return;

      document.getElementById('modal-content').innerHTML = `

        <span style="color: var(--gold); font-size: 12px; text-transform: uppercase; letter-spacing: 0.1em;">${p.category}</span>

        <h2 style="font-family: 'Cormorant Garamond', serif; font-size: 28px; margin-top: 6px;">${p.name}</h2>

        <div class="price" style="margin-top: 10px;">${p.price.toLocaleString('ru-RU')} ¥</div>

        <div class="modal-desc">${p.desc}</div>

        <button class="btn-primary" ${p.stock <= 0 ? 'disabled' : ''}
